cmake_minimum_required(VERSION 3.16)
project(misclab_webhooks CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(GTest REQUIRED)
add_library(misclab_verify src/webhook_signature.cpp)
target_include_directories(misclab_verify PUBLIC include)
target_link_libraries(misclab_verify PUBLIC OpenSSL::Crypto)
target_compile_options(misclab_verify PRIVATE -Wall -Wextra -Werror)
add_executable(misclab-verify src/main.cpp)
target_link_libraries(misclab-verify PRIVATE misclab_verify)
enable_testing()
add_executable(verify_tests test/webhook_signature_test.cpp)
target_link_libraries(verify_tests PRIVATE misclab_verify GTest::gtest_main)
target_compile_definitions(verify_tests PRIVATE VECTOR_PATH="${CMAKE_SOURCE_DIR}/../shared/test-vector.json")
include(GoogleTest)
gtest_discover_tests(verify_tests)
